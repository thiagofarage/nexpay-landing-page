import { Link, useNavigate } from "react-router-dom";
import { useState } from "react";

import { login } from "../services/auth";

const Login = () => {
  const navigate = useNavigate();
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const handleSubmit = async (event) => {
    event.preventDefault();
    if (isLoading) return;
    setError("");
    if (!email.trim() || !password) {
      setError("Preencha o email e a senha.");
      return;
    }
    setIsLoading(true);
    try {
      await login(email.trim(), password);
      setPassword("");
      navigate("/dashboard", { replace: true });
    } catch (error) {
      setError(error.message);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <main className="relative min-h-screen overflow-hidden bg-[#0B0B0B] text-white">
      {/* Glow de fundo */}
      <div className="pointer-events-none absolute left-1/2 top-0 h-125 w-125 -translate-x-1/2 rounded-full bg-[#4F8F83]/10 blur-[140px]" />

      {/* Voltar */}
      <Link
        to="/"
        className="absolute left-6 top-6 z-10 text-sm text-[#D9D9D9] transition-colors hover:text-white md:left-10 md:top-10"
      >
        ← Back to home
      </Link>

      <div className="relative z-10 flex min-h-screen items-center justify-center px-6 py-20">
        <div className="w-full max-w-md">
          {/* Marca */}
          <div className="mb-10 text-center">
            <div className="mx-auto mb-6 flex h-12 w-12 items-center justify-center rounded-xl border border-white/10 bg-[#1E1E1E]">
              <span className="text-lg font-bold">N</span>
            </div>

            <h1 className="text-3xl font-semibold tracking-tight">
              Welcome back
            </h1>

            <p className="mt-3 text-sm text-[#D9D9D9]/70">
              Sign in to continue with your Nexpay account.
            </p>
          </div>

          {/* Form */}
          <form
            onSubmit={handleSubmit}
            className="space-y-5"
            aria-busy={isLoading}
          >
            {error && (
              <p
                id="login-error"
                role="alert"
                className="rounded-xl border border-red-400/30 bg-red-400/10 p-3 text-sm text-red-200"
              >
                {error}
              </p>
            )}
            <div>
              <label
                htmlFor="email"
                className="mb-2 block text-sm font-medium text-[#D9D9D9]"
              >
                Email
              </label>

              <input
                id="email"
                name="email"
                required
                autoComplete="username"
                maxLength={150}
                disabled={isLoading}
                aria-describedby={error ? "login-error" : undefined}
                type="email"
                placeholder="name@company.com"
                value={email}
                onChange={(event) => setEmail(event.target.value)}
                className="h-12 w-full rounded-xl border border-white/10 bg-[#1E1E1E] px-4 text-sm text-white outline-none transition placeholder:text-white/30 focus:border-[#4F8F83]"
              />
            </div>

            <div>
              <div className="mb-2 flex items-center justify-between">
                <label
                  htmlFor="password"
                  className="text-sm font-medium text-[#D9D9D9]"
                >
                  Password
                </label>

                <button
                  type="button"
                  className="text-xs text-[#4F8F83] hover:underline"
                >
                  Forgot password?
                </button>
              </div>

              <input
                id="password"
                name="password"
                required
                autoComplete="current-password"
                disabled={isLoading}
                aria-describedby={error ? "login-error" : undefined}
                type="password"
                placeholder="••••••••"
                value={password}
                onChange={(event) => setPassword(event.target.value)}
                className="h-12 w-full rounded-xl border border-white/10 bg-[#1E1E1E] px-4 text-sm text-white outline-none transition placeholder:text-white/30 focus:border-[#4F8F83]"
              />
            </div>

            <div className="flex items-center gap-2">
              <input
                id="remember"
                disabled
                title="Disponível em uma próxima etapa"
                type="checkbox"
                className="h-4 w-4 accent-[#4F8F83]"
              />

              <label htmlFor="remember" className="text-sm text-[#D9D9D9]/70">
                Remember me
              </label>
            </div>

            <button
              type="submit"
              disabled={isLoading}
              className="disabled:cursor-wait disabled:opacity-60 cursor-pointer h-12 w-full rounded-xl bg-[#4F8F83] text-sm font-semibold text-[#0B0B0B] transition hover:bg-[#5fa396] active:scale-[0.99]"
            >
              {isLoading ? "Signing in..." : "Sign in"}
            </button>
          </form>

          <p className="mt-8 text-center text-sm text-[#D9D9D9]/60">
            Don't have an account?{" "}
            <button className="font-medium text-[#4F8F83] hover:underline">
              Create account
            </button>
          </p>
        </div>
      </div>
    </main>
  );
};

export default Login;
